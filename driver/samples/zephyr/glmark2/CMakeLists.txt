# glmark2: glmark2-es2 (Linaro's OpenGL ES 2.0 benchmark) as a Zephyr image on
# the board layer, one source for the Verdin AM62P, the Verdin iMX8M Plus and
# the Raspberry Pi 4 (driver/docs/glmark2.md).
#
#   -DGLMARK2_SRC=<glmark2 checkout at GLMARK2_COMMIT>
#
# glmark2 is GPL-3.0-or-later (its libmatrix is MIT). It is never vendored or
# committed here: its sources are compiled by path from the fetched tree. The
# one file this port changes, libmatrix/util.cc (MIT), is copied into the
# build directory and patched there (patches/). Everything else of upstream
# is compiled unchanged, including CanvasGeneric, GLStateEGL, the glad
# loaders and the bundled libpng / zlib / libjpeg-turbo. Ours: the native
# state over the board layer and the run (src/main.cpp), the dlopen/dlsym
# shim over eglGetProcAddress and the data-file lookup (src/platform.c).
cmake_minimum_required(VERSION 3.20.0)

get_filename_component(PVR_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/yakogl_dir.cmake")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(glmark2_zephyr)

# ---------------------------------------------------------------------------
# The upstream tree, at the pinned commit and unmodified.
# ---------------------------------------------------------------------------
set(GLMARK2_COMMIT 22c527cb0556f3a1ac4445aaa52cc532760928d5)
set(GLMARK2_VERSION "2023.01") # meson.build project version at that commit
if(NOT GLMARK2_SRC AND DEFINED ENV{GLMARK2_SRC})
  set(GLMARK2_SRC "$ENV{GLMARK2_SRC}")
endif()
if(NOT GLMARK2_SRC OR NOT EXISTS "${GLMARK2_SRC}/src/default-benchmarks.h")
  message(FATAL_ERROR
    "glmark2: GLMARK2_SRC='${GLMARK2_SRC}' is not a glmark2 checkout. Fetch it:\n"
    "  git clone https://github.com/glmark2/glmark2 <dir> && git -C <dir> checkout ${GLMARK2_COMMIT}\n"
    "and pass -DGLMARK2_SRC=<dir>.")
endif()
execute_process(COMMAND git -C "${GLMARK2_SRC}" rev-parse HEAD
  OUTPUT_VARIABLE _gm_head OUTPUT_STRIP_TRAILING_WHITESPACE
  ERROR_QUIET RESULT_VARIABLE _gm_rc)
if(NOT _gm_rc EQUAL 0)
  set(_gm_head "unknown")
endif()
execute_process(COMMAND git -C "${GLMARK2_SRC}" status --porcelain --untracked-files=no
  OUTPUT_VARIABLE _gm_dirty OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT _gm_head STREQUAL GLMARK2_COMMIT OR NOT _gm_dirty STREQUAL "")
  # A score names the code that produced it: another commit, or a modified
  # tree, is not the pinned glmark2.
  message(FATAL_ERROR
    "glmark2: ${GLMARK2_SRC} is at '${_gm_head}' (local changes: '${_gm_dirty}'), "
    "not the pinned ${GLMARK2_COMMIT} without local changes.")
endif()
set(GM "${GLMARK2_SRC}/src")
set(GM_DATA "${GLMARK2_SRC}/data")
set(GM_BUILD "${CMAKE_CURRENT_BINARY_DIR}/glmark2")
message(STATUS "glmark2: upstream ${GLMARK2_COMMIT} (${GLMARK2_VERSION}) from ${GLMARK2_SRC}")

include("${CMAKE_CURRENT_SOURCE_DIR}/../common/sample.cmake")

# ---------------------------------------------------------------------------
# libmatrix/util.cc: copied and patched (Zephyr: the data files come from the
# image, the way upstream's ANDROID arm takes them from the APK's assets).
# ---------------------------------------------------------------------------
set(_util_patch "${CMAKE_CURRENT_SOURCE_DIR}/patches/libmatrix-util-zephyr.patch")
set(_util_cc "${GM_BUILD}/src/libmatrix/util.cc")
find_program(PATCH patch REQUIRED)
add_custom_command(OUTPUT "${_util_cc}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${GM_BUILD}/src/libmatrix"
  COMMAND ${CMAKE_COMMAND} -E copy "${GM}/libmatrix/util.cc" "${_util_cc}"
  COMMAND ${PATCH} --batch --forward -p1 -d "${GM_BUILD}" -i "${_util_patch}"
  DEPENDS "${GM}/libmatrix/util.cc" "${_util_patch}"
  COMMENT "glmark2: patching libmatrix/util.cc for Zephyr")

# ---------------------------------------------------------------------------
# The data files (data/shaders, data/models, data/textures), linked in as
# they are and decoded on the board by upstream's own readers. One .S with an
# .incbin per file and a table of {name, data, size}; the names are the paths
# glmark2 asks for under GLMARK_DATA_PATH ("/glmark2").
# ---------------------------------------------------------------------------
file(GLOB_RECURSE _gm_data_files RELATIVE "${GM_DATA}" CONFIGURE_DEPENDS
     "${GM_DATA}/shaders/*" "${GM_DATA}/models/*" "${GM_DATA}/textures/*")
list(SORT _gm_data_files)
set(_asm "/* generated by glmark2/CMakeLists.txt: the glmark2 data files */\n")
string(APPEND _asm "\t.section .rodata.glmark2_data, \"a\"\n")
set(_names "\t.section .rodata.glmark2_names, \"a\"\n")
set(_tab "\t.section .rodata.glmark2_table, \"a\"\n\t.balign 8\n")
string(APPEND _tab "\t.global glmark2_asset_table\nglmark2_asset_table:\n")
set(_i 0)
set(_bytes 0)
set(_deps)
foreach(f ${_gm_data_files})
  file(SIZE "${GM_DATA}/${f}" _sz)
  math(EXPR _bytes "${_bytes} + ${_sz}")
  list(APPEND _deps "${GM_DATA}/${f}")
  string(APPEND _asm "\t.balign 16\n.Ld${_i}:\n\t.incbin \"${GM_DATA}/${f}\"\n.Ld${_i}_end:\n")
  string(APPEND _names ".Ln${_i}:\n\t.asciz \"/glmark2/${f}\"\n")
  string(APPEND _tab "\t.quad .Ln${_i}, .Ld${_i}, .Ld${_i}_end - .Ld${_i}\n")
  math(EXPR _i "${_i} + 1")
endforeach()
string(APPEND _tab "\t.global glmark2_asset_table_count\n\t.balign 8\n")
string(APPEND _tab "glmark2_asset_table_count:\n\t.quad ${_i}\n")
set(_assets_s "${CMAKE_CURRENT_BINARY_DIR}/glmark2_assets.S")
file(CONFIGURE OUTPUT "${_assets_s}" CONTENT "${_asm}${_names}${_tab}")
set_source_files_properties("${_assets_s}" PROPERTIES OBJECT_DEPENDS "${_deps}")
message(STATUS "glmark2: ${_i} data files, ${_bytes} bytes linked in")

# Upstream's code is built in static libraries of its own, with upstream's
# warnings (-w: not our warning set), and with the kernel's flags and headers
# through zephyr_interface: Zephyr's way for libraries of an application
# (zephyr_library() is for modules). The app links them; the linker pulls in
# what main.cpp references (SceneCollection registers every scene).
foreach(lib zlib png jpeg)
  add_library(glmark2_${lib} STATIC ${_${lib}})
  target_include_directories(glmark2_${lib} PRIVATE
    "${GM}/zlib" "${GM}/libpng" "${GM}/libjpeg-turbo")
  target_compile_options(glmark2_${lib} PRIVATE -w)
  target_link_libraries(glmark2_${lib} PUBLIC zephyr_interface)
  # kernel.h includes generated headers (heap_constants.h, ...)
  add_dependencies(glmark2_${lib} zephyr_generated_headers)
endforeach()
# zlib's gz* file functions use read/write/lseek/close: zlib's own configure
# defines Z_HAVE_UNISTD_H on a POSIX system.
target_compile_definitions(glmark2_zlib PRIVATE Z_HAVE_UNISTD_H)
target_compile_definitions(glmark2_png PRIVATE Z_HAVE_UNISTD_H)

add_library(glmark2_upstream STATIC ${_gm_cxx} ${_gm_glad} "${_util_cc}")
# YakoGL's include/ first: glad's EGL header takes <EGL/eglplatform.h> and
# <KHR/khrplatform.h> from there, so EGLNativeWindowType is YakoGL's
# struct gles_native_window * (glad's own eglplatform.h has no Zephyr arm).
# include/ holds the getopt.h, unistd.h and dlfcn.h glmark2 expects of a
# POSIX system (each says why).
target_include_directories(glmark2_upstream BEFORE PRIVATE
  "${YAKOGL_DIR}/include"
  "${GM}/glad/include" "${GM}" "${GM}/libmatrix" "${GM}/scene-ideas" "${GM}/scene-terrain"
  "${GM}/libpng" "${GM}/zlib" "${GM}/libjpeg-turbo"
  "${CMAKE_CURRENT_SOURCE_DIR}/include")
# _GNU_SOURCE: what g++ predefines on Linux, where glmark2 is built. Zephyr's
# POSIX layer sets _POSIX_C_SOURCE=200809L, which alone hides picolibc's XSI
# and BSD names (M_PI, used by the scenes).
target_compile_definitions(glmark2_upstream PRIVATE _GNU_SOURCE
  GLMARK2_USE_GLESv2 GLMARK2_USE_EGL USE_EXCEPTIONS
  GLMARK_VERSION="${GLMARK2_VERSION}" GLMARK_DATA_PATH="/glmark2"
  GLMARK2_EXECUTABLE="glmark2-es2")
target_compile_options(glmark2_upstream PRIVATE -w)
target_link_libraries(glmark2_upstream PUBLIC zephyr_interface
  glmark2_png glmark2_jpeg glmark2_zlib)
add_dependencies(glmark2_upstream zephyr_generated_headers)

  GLMARK2_USE_GLESv2 GLMARK2_USE_EGL USE_EXCEPTIONS
  GLMARK2_COMMIT="${GLMARK2_COMMIT}" GLMARK_VERSION="${GLMARK2_VERSION}")
target_link_libraries(app PRIVATE glmark2_upstream)
