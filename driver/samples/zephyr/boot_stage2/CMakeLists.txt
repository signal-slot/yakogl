cmake_minimum_required(VERSION 3.20.0)

# The second-stage loader (docs/notes/am62p-direct-boot.md, "The second
# stage"; docs/notes/imx8mp-direct-boot.md). Built for verdin_am62p/am62p54/a53
# or verdin_imx8mp/mimx8ml8/a53 with the board's overlay in boards/; the eMMC
# source adds emmc.conf and boards/<board>_emmc.overlay.
# The board support is in driver/ (a Zephyr module); the loader's logic is the
# Rust crate in loader/ (yakogl-boot-stage2), built as a staticlib through
# capi/yakogl_capi.cmake like the drivers archive.
get_filename_component(PVR_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${CMAKE_CURRENT_SOURCE_DIR}/../common/yakogl_dir.cmake")

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(boot_stage2)

if(NOT CONFIG_ARM64)
  message(FATAL_ERROR "boot_stage2: an aarch64 loader (verdin_am62p/am62p54/a53, verdin_imx8mp/mimx8ml8/a53)")
endif()

target_sources(app PRIVATE src/main.c)
target_include_directories(app PRIVATE loader/include)

# BOOT_STAGE2_SOURCE_EMBEDDED: -DBOOT_STAGE2_PAYLOAD=<stage2 .img> (made by
# tools/am62p-mkboot.py stage2img) is linked in as read-only data, header
# first, between bs2_payload_start and bs2_payload_end.
if(CONFIG_BOOT_STAGE2_SOURCE_EMBEDDED)
  if(NOT BOOT_STAGE2_PAYLOAD)
    message(FATAL_ERROR "boot_stage2: BOOT_STAGE2_SOURCE_EMBEDDED needs -DBOOT_STAGE2_PAYLOAD=<stage2 .img>")
  endif()
  get_filename_component(_bs2_payload "${BOOT_STAGE2_PAYLOAD}" ABSOLUTE)
  if(NOT EXISTS "${_bs2_payload}")
    message(FATAL_ERROR "boot_stage2: BOOT_STAGE2_PAYLOAD '${_bs2_payload}' does not exist")
  endif()
  set(_bs2_payload_s "${CMAKE_CURRENT_BINARY_DIR}/payload.S")
  file(WRITE "${_bs2_payload_s}"
"/* generated by boot_stage2/CMakeLists.txt: the stage2 image, header first */
	.section .rodata.bs2_payload, \"a\"
	.balign 64
	.global bs2_payload_start
bs2_payload_start:
	.incbin \"${_bs2_payload}\"
	.global bs2_payload_end
bs2_payload_end:
	.byte 0
")
  set_property(SOURCE "${_bs2_payload_s}" APPEND PROPERTY OBJECT_DEPENDS "${_bs2_payload}")
  target_sources(app PRIVATE "${_bs2_payload_s}")
elseif(BOOT_STAGE2_PAYLOAD)
  message(FATAL_ERROR "boot_stage2: BOOT_STAGE2_PAYLOAD given, but the source is not BOOT_STAGE2_SOURCE_EMBEDDED")
endif()

include("${YAKOGL_DIR}/capi/yakogl_capi.cmake")
set(_bs2_archive "${CMAKE_CURRENT_BINARY_DIR}/libboot_stage2.a")
yakogl_add_rust_archive(
  CRATE yakogl-boot-stage2
  MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/loader/Cargo.toml"
  LIB_NAME boot_stage2
  TARGET aarch64-unknown-none-softfloat
  RUSTUP_TOOLCHAIN "${YAKOGL_RUSTUP_TOOLCHAIN}"
  CARGO_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo-stage2"
  OUTPUT "${_bs2_archive}"
  TARGET_NAME boot_stage2_rust)
add_library(boot_stage2_rust_lib STATIC IMPORTED GLOBAL)
set_target_properties(boot_stage2_rust_lib PROPERTIES IMPORTED_LOCATION "${_bs2_archive}")
add_dependencies(boot_stage2_rust_lib boot_stage2_rust)
add_dependencies(app boot_stage2_rust)
target_link_libraries(app PRIVATE boot_stage2_rust_lib)
