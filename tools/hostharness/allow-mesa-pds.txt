# Only for comparing a build before source-form step 3 (Mesa's PDS generators
# and csbgen; main up to land123) with one after it (land124 and later):
#   HH_ALLOW=tools/hostharness/allow-mesa-pds.txt tools/hostharness/compare.sh ...
#
# The one class of draws whose command stream changed: a vertex shader that
# reads gl_VertexID or gl_InstanceID. Mesa's pvr_pds_vertex_shader keeps the
# two id DOUTW control words in spare constants of the stream blocks, so the
# vertex PDS data segment is 8 dwords smaller (VDM_STATE5 vs_pds_data_size
# e.g. 8 -> 6 units) and the DOUTW src1 indices move; the instructions are
# otherwise the same. Two smaller changes ride along: a vertex program with no
# stream and no id declares its 2 data dwords (the old encoder declared 0
# while its DOUTU read constant 0), and the clear vertex program writes 0 into
# its unused multiplier constant. The jobs below are the gl_test cases that
# draw with such a shader (43 44 45 47 49 51 57 141), the gl_test suites that
# contain them and the Quick 3D programs.
#
# What differs in those logs: the per-draw dump lines vpds_data / vpds_code /
# vdm, the hashes of the memory that holds them, and, because the shorter
# data segment dumps fewer lines, the numbering of every later log line (and
# the log count at hh-end). Shorter dumps make hunks of unequal length, which
# only `line` rules can explain, so the log rule covers any library log line
# in these jobs. Checked when the rules were written (land124, dev and
# release, every mode, 288 logs): every line that is not a library log line
# or a memory hash (wrapper calls and results, submits and their kick hashes,
# fences, cache maintenance, gl_test's output) is identical and in the same
# order on both sides, and the log lines that differ as a multiset are only
# vpds_data, vpds_code and vdm (plus host-pointer, allow.txt). Where diff(1)
# aligns the long gltest.51 logs differently (l3wf), identical wrapper and
# submit lines land in the hunks: pds-vid-drawcall and pds-vid-align51.

pds-vid-renumber	^(gltest\.(43|44|45|47|49|51|57|141|all|all-dsdump|all-pretransform)|scen\.progs-qt3d)$	sub	^log #\d+([:+]) => log #N\1	log lines renumbered after a shorter vertex PDS dump (Mesa's vertex-id layout)
pds-vid-log	^(gltest\.(43|44|45|47|49|51|57|141|all|all-dsdump|all-pretransform)|scen\.progs-qt3d)$	line	^log #\d+[:+] 	library log lines in hunks of unequal length: the vpds_data/vpds_code/vdm dump of an id draw and the renumbered lines after it
pds-vid-mem	^(gltest\.(43|44|45|47|49|51|57|141|all|all-dsdump|all-pretransform)|scen\.progs-qt3d)$	line	^  (mem #\d+ gpu=[0-9a-f]+ size=\d+ hash=|memhash\[)	hashes of the memory holding the vertex PDS program and the VDM stream
pds-vid-logcount	^(gltest\.(43|44|45|47|49|51|57|141|all|all-dsdump|all-pretransform)|scen\.progs-qt3d)$	sub	^(hh-end .*) logs=\d+ => \1 logs=N	the log line count at hh-end (shorter dumps)
pds-vid-drawcall	^(gltest\.(43|44|45|47|49|51|57|141|all|all-dsdump|all-pretransform)|scen\.progs-qt3d)$	line	^~glDraw(Arrays|Elements)(Instanced)?\(	identical draw-call lines that diff(1) puts into a renumbered hunk (the program's own arguments)
pds-vid-align51	^gltest\.51$	line	^(submit #\d+ len=|  kick |  -> fence |cache_flush #\d+ |completed -> |wait \d+ t=|gl_test: heavy: |~gl(Clear|ClearColor|Finish)\(|hh-end )	gltest.51 l3wf: diff(1) aligns one identical submit block differently on the two sides (same lines, same order)
