# femtovg: one femtovg frame (the 2D renderer Slint draws with) for the
# Raspberry Pi 4, the Verdin AM62P and the Verdin iMX8M Plus. The board is
# chosen by the Zephyr board (-b) only; there is no per-board option here.
# What differs per board is the board layer's (driver/zephyr/lib/board,
# CONFIG_YAKOGL_BOARD, and its app.conf and app.overlay, which
# ../zephyr.cmake applies).
#
# What is shared, by path:
#   - the femtovg/glow forks (NOSTD_PATCH_DIR), recreated from ../nostd's
#     patches (../nostd/README.md), the same trees the Slint example uses;
#   - femtovg's own shader text (main-vs.glsl, main-fs.glsl) from that tree,
#     which src/programs.c compiles the way femtovg does, to name each
#     program that does not build before the frame;
#   - the scene and the check, ./check (crate yakogl-femtovg-check,
#     host-tested by tools/check.sh), and through it ../triangle/check and
#     driver/rust/src/board/record.rs.
cmake_minimum_required(VERSION 3.20.0)

include("${CMAKE_CURRENT_SOURCE_DIR}/../zephyr.cmake")
project(femtovg)

target_sources(app PRIVATE src/main.c src/programs.c)

# ---------------------------------------------------------------------------
# The femtovg/glow forks.
# ---------------------------------------------------------------------------
if(NOT NOSTD_PATCH_DIR AND DEFINED ENV{NOSTD_PATCH_DIR})
  set(NOSTD_PATCH_DIR "$ENV{NOSTD_PATCH_DIR}")
endif()
if(NOT EXISTS "${NOSTD_PATCH_DIR}/femtovg/Cargo.toml" OR NOT EXISTS "${NOSTD_PATCH_DIR}/glow/Cargo.toml")
  message(FATAL_ERROR
    "femtovg: NOSTD_PATCH_DIR='${NOSTD_PATCH_DIR}' must hold the no_std femtovg/ and "
    "glow/ forks (../nostd/README.md). Pass -DNOSTD_PATCH_DIR=<path>.")
endif()

# femtovg's shader text as C strings (gen/fvg_glsl.h: fvg_main_vs,
# fvg_main_fs), from the tree the staticlib builds femtovg from, so the
# survey's programs are the ones femtovg will build.
set(fvg_glsl_dir "${NOSTD_PATCH_DIR}/femtovg/src/renderer/opengl")
set(fvg_glsl_h "${CMAKE_CURRENT_BINARY_DIR}/gen/fvg_glsl.h")
set(fvg_glsl "/* generated by examples/femtovg/CMakeLists.txt from ${fvg_glsl_dir} */\n")
foreach(pair "main-vs.glsl;fvg_main_vs" "main-fs.glsl;fvg_main_fs")
  list(GET pair 0 file)
  list(GET pair 1 sym)
  if(NOT EXISTS "${fvg_glsl_dir}/${file}")
    message(FATAL_ERROR "femtovg: ${fvg_glsl_dir}/${file} is missing")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${fvg_glsl_dir}/${file}")
  file(READ "${fvg_glsl_dir}/${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  string(APPEND fvg_glsl "static const char ${sym}[] = { ${bytes} 0 };\n")
endforeach()
file(WRITE "${fvg_glsl_h}.tmp" "${fvg_glsl}")
configure_file("${fvg_glsl_h}.tmp" "${fvg_glsl_h}" COPYONLY)
target_include_directories(app PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/gen")

# ---------------------------------------------------------------------------
# The Rust staticlib (rust/): built for aarch64-unknown-none (release) with
# -Z build-std=core,alloc, as the Slint example's is, and linked whole.
# ---------------------------------------------------------------------------
find_program(CARGO cargo REQUIRED)
set(CARGO_SRC "${CMAKE_CURRENT_SOURCE_DIR}/rust")
set(FEMTOVG_CHECK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/check")
set(cargo_build_dir "${CMAKE_CURRENT_BINARY_DIR}/rust")
set(cargo_target_dir "${cargo_build_dir}/target")
set(RUST_TARGET aarch64-unknown-none)
set(femtovg_staticlib "${cargo_target_dir}/${RUST_TARGET}/release/libfemtovg_example.a")
file(MAKE_DIRECTORY "${cargo_build_dir}")
configure_file("${CARGO_SRC}/Cargo.toml.in" "${cargo_build_dir}/Cargo.toml" @ONLY)

# The crates.io part of the build (what femtovg and glow pull in) is pinned by
# rust/Cargo.lock, used with --locked, as the Slint example's is.
# -DFEMTOVG_CARGO_LOCKED=OFF resolves afresh, to refresh the lock on purpose
# (copy <build>/rust/Cargo.lock over rust/Cargo.lock and commit it).
option(FEMTOVG_CARGO_LOCKED "Build with the committed rust/Cargo.lock and --locked" ON)
set(femtovg_lock "${CARGO_SRC}/Cargo.lock")
set(femtovg_cargo_flags "")
set(femtovg_lock_dep "")
if(FEMTOVG_CARGO_LOCKED)
  configure_file("${femtovg_lock}" "${cargo_build_dir}/Cargo.lock" COPYONLY)
  set(femtovg_cargo_flags --locked)
  set(femtovg_lock_dep "${femtovg_lock}")
endif()

file(GLOB_RECURSE femtovg_rust_srcs CONFIGURE_DEPENDS
     "${CARGO_SRC}/src/*.rs"
     "${FEMTOVG_CHECK_DIR}/src/*.rs"
     "${CMAKE_CURRENT_SOURCE_DIR}/../triangle/check/src/*.rs"
     "${PVR_REPO}/rust/src/board/record.rs"
     "${NOSTD_PATCH_DIR}/femtovg/src/*.rs"
     "${NOSTD_PATCH_DIR}/femtovg/src/*.glsl"
     "${NOSTD_PATCH_DIR}/glow/src/*.rs")

add_custom_command(
  OUTPUT "${femtovg_staticlib}"
  COMMAND ${CMAKE_COMMAND} -E env
          "CARGO_TARGET_DIR=${cargo_target_dir}"
          ${CARGO} build --release ${femtovg_cargo_flags}
          -Z build-std=core,alloc
          --target ${RUST_TARGET}
          --manifest-path "${cargo_build_dir}/Cargo.toml"
  DEPENDS ${femtovg_rust_srcs} "${cargo_build_dir}/Cargo.toml" ${femtovg_lock_dep}
  WORKING_DIRECTORY "${cargo_build_dir}"
  COMMENT "cargo build femtovg_example (${RUST_TARGET}, release)"
  VERBATIM)
add_custom_target(femtovg_example_build DEPENDS "${femtovg_staticlib}")
add_dependencies(app femtovg_example_build)
target_link_libraries(app PRIVATE "${femtovg_staticlib}")
